How it works

Bigram "statistics" → "helps" is stored as a path in the tree:

ROOT
↓ (follow "statistics")
"statistics" (is_end = True)
↓ (read next_words)
next_words: { "helps": 3, "involves": 1, "and": 2 }

Trigram "data science" → traverse two levels:
root → "data" → "science" → next_words

ROOT
↓
"data"
↓
"science" (is_end = True)
↓
next_words: { "involves": 2, "is": 1 }

🗄️ SQLite Schema & Queries

Tables & Indexes 
-- N-gram counts stored in 3 tables

CREATE TABLE unigrams (
  word  TEXT PRIMARY KEY,
  count INTEGER
);

CREATE TABLE bigrams (
  w1    TEXT NOT NULL,
  w2    TEXT NOT NULL,
  count INTEGER,
  PRIMARY KEY (w1, w2)
);
CREATE INDEX idx_bigrams_w1
  ON bigrams (w1);

CREATE TABLE trigrams (
  w1 TEXT, w2 TEXT, w3 TEXT,
  count INTEGER,
  PRIMARY KEY (w1, w2, w3)
);
CREATE INDEX idx_trigrams_w1w2
  ON trigrams (w1, w2);

-- Query: What comes after "statistics"?
SELECT w2, count
FROM   bigrams
WHERE  w1 = 'statistics'
ORDER  BY count DESC;
